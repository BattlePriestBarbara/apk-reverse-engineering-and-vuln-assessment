 236|     return SIGN_ALGORITHMS[algorithm](payload, secret)

 241|     message = f"{canonical_query(payload)}{secret}".encode("utf-8")
 242|     return hmac.new(secret.encode("utf-8"), message,

 271| def enforce_guard(cfg: PocConfig) -> None:
 272|     """执行守卫检查，任何一项不通过即中止。
 273| 
 274|     Raises
 275|     ------
 276|     GuardError
 277|         指向生产域名、主机不在白名单、配置仍是占位符或 scheme 非法。
 278|     """
 279|     if cfg.secret in PLACEHOLDER_VALUES:
 280|         raise GuardError("secret 仍是占位符，请用 --secret-stdin 注入")
 281|     if cfg.app_key in PLACEHOLDER_VALUES:
 282|         raise GuardError("appKey 仍是占位符，请用 --app-key 传入")
 283|     host = cfg.host()
 284|     if not host:
 285|         raise GuardError(f"无法解析主机名：{cfg.url!r}")
 286|     if host in PLACEHOLDER_VALUES or "your_" in host:
 287|         raise GuardError(f"目标主机仍是占位符：{host!r}，请显式 --url")
 288|     for prod in PRODUCTION_HOSTS:
 289|         if host == prod or host.endswith("." + prod):
 290|             raise GuardError(f"目标 {host} 属生产域名，本脚本拒绝访问")
 291|     allowed = cfg.allow_hosts or (host,)
 292|     if host not in allowed:
 293|         raise GuardError(f"主机 {host} 不在 --allow-host 白名单 {allowed}")
 294|     if urlsplit(cfg.url).scheme not in ("http", "https"):
 295|         raise GuardError("仅支持 http/https")
